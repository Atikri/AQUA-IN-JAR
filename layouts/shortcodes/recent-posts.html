{{/* Shortcode: recent-posts Usage: {{< recent-posts count=4>}} Renders a list of latest posts by date (desc), default 4 items. */}} {{- $count := .Get "count" | default 4 -}} {{- $all := site.RegularPages -}} {{- $published := where $all "Draft" "ne" true -}} {{/* Filter out secret-chamber content */}} {{- $filtered := slice -}} {{- range $published -}} {{- if not (in .File.Dir "secret-chamber") -}} {{- $filtered = $filtered | append . -}} {{- end -}} {{- end -}} {{- $sorted := sort $filtered "Date" "desc" -}} {{- $items := first $count $sorted -}}
{{- range $items }}
{{ .Title }}
{{ .Date.Format "2006-01-02" }}
 {{- end }}
